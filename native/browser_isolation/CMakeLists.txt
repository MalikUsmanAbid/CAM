cmake_minimum_required(VERSION 3.24)
project(jarvis_browser_isolation LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "Windows SDK and MSVC are required")
endif()
add_library(jarvis_browser_fence STATIC browser_fence.cpp managed_deployment.cpp elevation_bridge.cpp)
target_compile_features(jarvis_browser_fence PUBLIC cxx_std_20)
target_compile_definitions(jarvis_browser_fence PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(jarvis_browser_fence PUBLIC Fwpuclnt Rpcrt4 Advapi32 Userenv Ole32 Shell32 Uuid)
target_include_directories(jarvis_browser_fence PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(jarvis-isolation-probe-host probe_host.cpp)
target_link_libraries(jarvis-isolation-probe-host PRIVATE jarvis_browser_fence)
add_executable(jarvis-network-probe network_probe.cpp)
add_executable(jarvis-browser-recovery-inventory recovery_inventory.cpp)
target_link_libraries(jarvis-browser-recovery-inventory PRIVATE jarvis_browser_fence)
target_compile_features(jarvis-network-probe PRIVATE cxx_std_20)
target_compile_definitions(jarvis-network-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(jarvis-network-probe PRIVATE Ws2_32)
